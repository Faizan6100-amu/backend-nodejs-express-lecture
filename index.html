Module 1: Introduction to Node.js

Node.js is an open-source, cross-platform JavaScript runtime environment built on Chrome's V8 JavaScript engine that allows you to execute JavaScript outside of a browser.

What is Node.js?

Node.js enables developers to use JavaScript for server-side development with the following key features:

  • Event-Driven Architecture: Non-blocking, asynchronous I/O operations
  • Single-Threaded: Uses event loop for handling concurrent requests
  • Fast Execution: Built on V8 engine (same as Chrome)
  • NPM Ecosystem: Access to millions of reusable packages
  • Scalability: Excellent for I/O-heavy applications
  • Cross-Platform: Works on Windows, Linux, macOS

Node.js Architecture

JavaScript Code (Single-threaded)
Event Loop (Manages async callbacks)
libuv (Asynchronous I/O)
V8 Engine (JS Execution)
OS System Calls

Module 3: Event-Driven Programming

Understanding Events

Event-driven programming is a paradigm where the flow of the program is determined by events. In Node.js, events are handled using the EventEmitter class.

Basic Event Emitter

const EventEmitter = require('events');

class MyEmitter extends EventEmitter {}

const emitter = new MyEmitter();

// Register event listener
emitter.on('userLoggedIn', (user) => {
    console.log(`${user} logged in successfully!`);
});

// Emit event
emitter.emit('userLoggedIn', 'Alice');
// Output: Alice logged in successfully!

Timers: setTimeout, setInterval, setImmediate

// setTimeout - Executes after delay (once)
setTimeout(() => {
    console.log('This runs after 2 seconds');
}, 2000);

// setInterval - Executes repeatedly
let count = 0;
const intervalId = setInterval(() => {
    count++;
    console.log(`Counter: ${count}`);
    if (count >= 5) clearInterval(intervalId);
}, 1000);

// setImmediate - Next iteration of event loop
setImmediate(() => {
    console.log('Immediate callback');
});

Callbacks - Error-First Convention

const fs = require('fs');

// Error-first callback pattern
fs.readFile('data.txt', 'utf8', (error, data) => {
    if (error) {
        console.error('Error reading file:', error);
        return;
    }
    console.log('File content:', data);
});

// Custom callback function
function fetchUserData(userId, callback) {
    setTimeout(() => {
        const user = { id: userId, name: 'John Doe' };
        callback(null, user); // null = no error
    }, 1000);
}

fetchUserData(1, (error, user) => {
    if (error) console.error(error);
    else console.log('User:', user);
});

Handling Callback Hell with Async/Await

// Before: Callback Hell
fs.readFile('file1.txt', 'utf8', (err, data1) => {
    if (err) throw err;
    fs.readFile('file2.txt', 'utf8', (err, data2) => {
        if (err) throw err;
        console.log(data1, data2);
    });
});

// After: Using async/await
async function readFiles() {
    try {
        const data1 = await fs.promises.readFile('file1.txt', 'utf8');
        const data2 = await fs.promises.readFile('file2.txt', 'utf8');
        console.log(data1, data2);
    } catch (error) {
        console.error('Error:', error);
    }
}

readFiles();

Directory Operations

const fs = require('fs').promises;
const path = require('path');

async function manageDirs() {
    try {
        // Create directory
        await fs.mkdir('my-folder', { recursive: true });

        console.error('Error:', error);
    }
}

Practical: Recursive Directory Reader

async function readDirRecursive(dirPath) {
    try {
        const files = await fs.readdir(dirPath, { withFileTypes: true });

        for (const file of files) {
            const fullPath = path.join(dirPath, file.name);

            if (file.isDirectory()) {
                console.log(`[DIR] ${fullPath}`);
                await readDirRecursive(fullPath);
            } else {
                console.log(`[FILE] ${fullPath}`);
            }
        }
    } catch (error) {
        console.error('Error:', error);
    }
}

Module 5: Express.js Framework

Introduction to Express.js

Express.js is a minimal and flexible Node.js web application framework that provides a robust set of features for building web applications and APIs.

Key Features:
  • Minimalist: Lightweight and unopinionated
  • Routing: Powerful routing system
  • Middleware: Flexible middleware system
  • Performance: Fast and efficient
  • Scalable: Suitable for small to large applications

Middleware

// Logging middleware
app.use((req, res, next) => {
    console.log(`${req.method} ${req.url}`);
    next(); // Pass to next middleware
});

// Authentication middleware
const authenticate = (req, res, next) => {
    const token = req.headers.authorization;
    if (token === 'Bearer valid-token') {
        next();
    } else {
        res.status(401).json({ error: 'Unauthorized' });
    }
};

app.get('/protected', authenticate, (req, res) => {
    res.json({ message: 'Protected resource' });
});

// Error handling middleware (must have 4 parameters)
app.use((err, req, res, next) => {
    console.error(err.message);
    res.status(500).json({ error: err.message });
});

What is REST?

REST (Representational State Transfer) is an architectural style for designing networked applications using standard HTTP methods.

HTTP Methods in REST:

Complete To-Do API Example

const express = require('express');
const app = express();

app.use(express.json());

// In-memory data store
let todos = [];
let nextId = 1;

// GET all todos
app.get('/api/todos', (req, res) => {
    res.json(todos);
});

// GET specific todo
app.get('/api/todos/:id', (req, res) => {
    const todo = todos.find(t => t.id === parseInt(req.params.id));
    if (!todo) {
        return res.status(404).json({ error: 'Todo not found' });
    }
    res.json(todo);
});

// CREATE new todo
app.post('/api/todos', (req, res) => {
    if (!req.body.title) {
        return res.status(400).json({ error: 'Title is required' });
    }
    
    const newTodo = {
        id: nextId++,
        title: req.body.title,
        completed: false
    };
    
    todos.push(newTodo);
    res.status(201).json(newTodo);
});

// UPDATE todo
app.put('/api/todos/:id', (req, res) => {
    const todo = todos.find(t => t.id === parseInt(req.params.id));
    if (!todo) {
        return res.status(404).json({ error: 'Todo not found' });
    }
    
    if (req.body.title) todo.title = req.body.title;
    if (req.body.completed !== undefined) todo.completed = req.body.completed;
    
    res.json(todo);
});

// DELETE todo
app.delete('/api/todos/:id', (req, res) => {
    const index = todos.findIndex(t => t.id === parseInt(req.params.id));
    if (index === -1) {
        return res.status(404).json({ error: 'Todo not found' });
    }
    
    const deleted = todos.splice(index, 1);
    res.json(deleted[0]);
});

app.listen(3000, () => {
    console.log('API running on port 3000');
});

User Management System

const express = require('express');
const fs = require('fs').promises;
const path = require('path');

const app = express();
app.use(express.json());

const usersFile = path.join(__dirname, 'users.json');

// Get all users
app.get('/api/users', async (req, res) => {
    try {
        const data = await fs.readFile(usersFile, 'utf8');
        const users = JSON.parse(data);
        res.json(users);
    } catch (error) {
        res.status(500).json({ error: 'Failed to fetch users' });
    }
});

// Create user
app.post('/api/users', async (req, res) => {
    try {
        const { name, email } = req.body;
        
        if (!name || !email) {
            return res.status(400).json({ error: 'Name and email required' });
        }
        
        const data = await fs.readFile(usersFile, 'utf8');
        const users = JSON.parse(data);
        
        const newUser = {
            id: users.length + 1,
            name,
            email,
            createdAt: new Date()
        };
        
        users.push(newUser);
        await fs.writeFile(usersFile, JSON.stringify(users, null, 2));
        
        res.status(201).json(newUser);
    } catch (error) {
        res.status(500).json({ error: 'Failed to create user' });
    }
});

app.listen(3000, () => {
    console.log('Server running on port 3000');
});

Task Scheduler with Events

const express = require('express');
const EventEmitter = require('events');

const app = express();
app.use(express.json());

class TaskManager extends EventEmitter {
    constructor() {
        super();
        this.tasks = [];
        this.nextId = 1;
    }
    
    addTask(title, description, dueDate) {
        const task = {
            id: this.nextId++,
            title,
            description,
            dueDate: new Date(dueDate),
            status: 'pending',
            createdAt: new Date()
        };
        
        this.tasks.push(task);
        this.emit('taskCreated', task);
        return task;
    }
    
    completeTask(taskId) {
        const task = this.tasks.find(t => t.id === taskId);
        if (!task) return null;
        
        task.status = 'completed';
        task.completedAt = new Date();
        this.emit('taskCompleted', task);
        return task;
    }
}

const taskManager = new TaskManager();

// Event listeners
taskManager.on('taskCreated', (task) => {
    console.log(`📌 New task: ${task.title}`);
});

taskManager.on('taskCompleted', (task) => {
    console.log(`✅ Completed: ${task.title}`);
});

// Routes
app.post('/api/tasks', (req, res) => {
    const { title, description, dueDate } = req.body;
    const task = taskManager.addTask(title, description, dueDate);
    res.status(201).json(task);
});

app.get('/api/tasks', (req, res) => {
    res.json(taskManager.tasks);
});

app.put('/api/tasks/:id/complete', (req, res) => {
    const task = taskManager.completeTask(parseInt(req.params.id));
    if (!task) {
        return res.status(404).json({ error: 'Task not found' });
    }
    res.json(task);
});

app.listen(3000, () => {
    console.log('Task Manager running on port 3000');
});

Complete Blog API

const express = require('express');
const app = express();

app.use(express.json());

// Database simulation
const db = {
    posts: [
        {
            id: 1,
            title: 'Getting Started with Node.js',
            content: 'Node.js is a runtime environment...',
            author: 'John Doe',
            createdAt: new Date('2024-01-01')
        }
    ],
    nextPostId: 2
};

// Middleware: validation
const validatePost = (req, res, next) => {
    const { title, content, author } = req.body;
    
    if (!title || !content || !author) {
        return res.status(400).json({
            error: 'Missing required fields'
        });
    }
    
    if (title.length < 5) {
        return res.status(400).json({
            error: 'Title must be at least 5 characters'
        });
    }
    
    next();
};

// Routes
app.get('/api/posts', (req, res) => {
    const { sort } = req.query;
    let posts = [...db.posts];
    
    if (sort === 'newest') {
        posts.sort((a, b) => b.createdAt - a.createdAt);
    }
    
    res.json({
        success: true,
        data: posts
    });
});

app.post('/api/posts', validatePost, (req, res) => {
    const { title, content, author } = req.body;
    
    const newPost = {
        id: db.nextPostId++,
        title,
        content,
        author,
        createdAt: new Date()
    };
    
    db.posts.push(newPost);
    
    res.status(201).json({
        success: true,
        message: 'Post created',
        data: newPost
    });
});

app.delete('/api/posts/:id', (req, res) => {
    const index = db.posts.findIndex(p => p.id === parseInt(req.params.id));
    
    if (index === -1) {
        return res.status(404).json({ error: 'Post not found' });
    }
    
    const deleted = db.posts.splice(index, 1);
    res.json({ success: true, data: deleted[0] });
});

app.listen(3000, () => {
    console.log('Blog API running on port 3000');
});
